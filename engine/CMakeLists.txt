target_include_directories(heartstead_engine PUBLIC ${PROJECT_SOURCE_DIR})
target_link_libraries(heartstead_engine PUBLIC Threads::Threads PRIVATE fastgltf::fastgltf)
if(HEARTSTEAD_ENABLE_MINIAUDIO)
  target_sources(heartstead_engine PRIVATE audio/miniaudio/miniaudio_implementation.cpp)
  target_include_directories(heartstead_engine SYSTEM PRIVATE ${MINIAUDIO_INCLUDE_DIRS})
  target_link_libraries(heartstead_engine PRIVATE ${CMAKE_DL_LIBS})
  target_compile_definitions(heartstead_engine PRIVATE HEARTSTEAD_HAS_MINIAUDIO=1)
else()
  target_compile_definitions(heartstead_engine PRIVATE HEARTSTEAD_HAS_MINIAUDIO=0)
endif()
if(HEARTSTEAD_ENABLE_JOLT)
  target_link_libraries(heartstead_engine PRIVATE Jolt::Jolt)
  target_compile_definitions(heartstead_engine PRIVATE HEARTSTEAD_HAS_JOLT=1)
  if(HEARTSTEAD_SANITIZERS MATCHES "(^|,)thread(,|$)")
    set_property(
      SOURCE physics/jolt/jolt_backend.cpp
      APPEND
      PROPERTY COMPILE_DEFINITIONS HEARTSTEAD_JOLT_SINGLE_THREADED=1
    )
  else()
    set_property(
      SOURCE physics/jolt/jolt_backend.cpp
      APPEND
      PROPERTY COMPILE_DEFINITIONS HEARTSTEAD_JOLT_SINGLE_THREADED=0
    )
  endif()
  if(HEARTSTEAD_SANITIZERS MATCHES "(^|,)undefined(,|$)")
    # The vcpkg Jolt binary is built without C++ RTTI. Clang's vptr sanitizer otherwise emits
    # typeinfo references for Jolt classes that cannot be resolved at link time.
    set_property(
      SOURCE physics/jolt/jolt_backend.cpp
      APPEND
      PROPERTY COMPILE_OPTIONS -fno-sanitize=vptr
    )
  endif()
else()
  target_compile_definitions(heartstead_engine PRIVATE HEARTSTEAD_HAS_JOLT=0)
endif()
if(HEARTSTEAD_ENABLE_VULKAN AND Vulkan_FOUND)
  target_link_libraries(heartstead_engine PRIVATE Vulkan::Vulkan)
  target_compile_definitions(heartstead_engine PRIVATE HEARTSTEAD_HAS_VULKAN=1)
else()
  target_compile_definitions(heartstead_engine PRIVATE HEARTSTEAD_HAS_VULKAN=0)
endif()
if(HEARTSTEAD_ENABLE_X11_NATIVE AND X11_FOUND)
  target_include_directories(heartstead_engine PRIVATE ${X11_INCLUDE_DIR})
  target_link_libraries(heartstead_engine PRIVATE ${X11_LIBRARIES})
  target_compile_definitions(heartstead_engine PRIVATE HEARTSTEAD_HAS_X11=1)
  if(HEARTSTEAD_ENABLE_XRANDR AND X11_Xrandr_FOUND)
    if(TARGET X11::Xrandr)
      target_link_libraries(heartstead_engine PRIVATE X11::Xrandr)
    else()
      target_link_libraries(heartstead_engine PRIVATE ${X11_Xrandr_LIB})
      target_include_directories(heartstead_engine PRIVATE ${X11_Xrandr_INCLUDE_PATH})
    endif()
    target_compile_definitions(heartstead_engine PRIVATE HEARTSTEAD_HAS_XRANDR=1)
  else()
    target_compile_definitions(heartstead_engine PRIVATE HEARTSTEAD_HAS_XRANDR=0)
  endif()
else()
  target_compile_definitions(heartstead_engine PRIVATE HEARTSTEAD_HAS_X11=0)
  target_compile_definitions(heartstead_engine PRIVATE HEARTSTEAD_HAS_XRANDR=0)
endif()

heartstead_configure_target(heartstead_engine)
